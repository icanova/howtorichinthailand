โหมดทดลอง — เล่นได้เฉพาะในเบราว์เซอร์นี้ (เปิดหลายแท็บเพื่อลองหลายคน) · ใส่คีย์ Supabase เพื่อเล่นข้ามเครื่อง
เชื่อมต่อ Supabase (เล่นข้ามเครื่องจริง)

เปิดไฟล์นี้ แก้ 2 บรรทัดบนสุดของ <script> ท้ายไฟล์:

const SUPABASE_URL  = '...';
const SUPABASE_ANON = '...';

แล้วรัน SQL นี้ใน Supabase > SQL Editor (สำหรับ lobby):

create table rooms (
  id uuid primary key default gen_random_uuid(),
  code text unique not null,
  status text not null default 'lobby',
  host_identity text not null,
  round_number int not null default 0,
  current_phase int not null default 1,
  created_at timestamptz default now()
);
create table players (
  id uuid primary key default gen_random_uuid(),
  room_id uuid references rooms(id) on delete cascade,
  identity text not null,
  display_name text not null,
  color text not null,
  seat_order int not null,
  is_connected boolean default true,
  joined_at timestamptz default now()
);
alter publication supabase_realtime add table rooms;
alter publication supabase_realtime add table players;

-- MVP: เปิดกว้างไว้ก่อน แล้วค่อยรัด RLS ทีหลัง
alter table rooms   enable row level security;
alter table players enable row level security;
create policy p_rooms   on rooms   for all using (true) with check (true);
create policy p_players on players for all using (true) with check (true);
